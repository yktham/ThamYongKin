Bonus Question:
// Your code here
const p = Promise.resolve('World');
// Log the initial promise (pending state)
console.log(p); // Promise { <pending> }

// Chain another async operation
p
  .then((value) => {
    // Return a new promise that resolves after 1 second
    return new Promise((resolve) => {
      setTimeout(() => {
        resolve('Hello ' + value);
      }, 1000);
    });
  })
  .then((finalValue) => {
    console.log(finalValue); // Logs "Hello World" after 1 second
  });
